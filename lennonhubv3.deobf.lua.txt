				Headers = { ["User-Agent"] = "AuraHub-RobloxClient/2.0" },
			})
		end)

			local statusCode = res.StatusCode or res.status_code or 200

local function SafeHttpRequest(url)
	local ok, body, code = RawHttpRequest(url)
	local isCloudflare = false
	if ok and body then
		if
			string.find(body, "challenges.cloudflare.com")
			or string.find(body, "<title>Just a moment...")
			or string.find(body, "cf%-mitigated")
			or (code == 403 and string.find(body, "<html"))
		then
			isCloudflare = true
		end
	elseif not ok or code == 403 then
		isCloudflare = true
	end
	if
		isCloudflare
		and BACKUP_SERVER_URL
		and BACKUP_SERVER_URL ~= ""
		and PRIMARY_SERVER_URL
		and PRIMARY_SERVER_URL ~= ""
	then
		local backupUrl = string.gsub(url, PRIMARY_SERVER_URL, BACKUP_SERVER_URL)
		if backupUrl ~= url then
			local bOk, bBody, bCode = RawHttpRequest(backupUrl)
			if bOk and bBody and not string.find(bBody, "<title>Just a moment...") then
				return bOk, bBody, bCode
			end
		end
	end
	return ok, body, code
end

local function ParseSavedKeyForScript(fileContent, targetScript)
	if not fileContent or fileContent == "" then
		return nil
	end
	local target = string.lower(string.gsub(targetScript or "", "^%s*(.-)%s*$", "%1"))
	local singleKeyFallback = nil
	for line in string.gmatch(fileContent, "[^\r\n]+") do
		local trimmed = string.gsub(line, "^%s*(.-)%s*$", "%1")
		if
			trimmed ~= ""
			and string.sub(trimmed, 1, 2) ~= "--"
			and string.sub(trimmed, 1, 1) ~= "#"
		then
			local cleanLine = string.gsub(trimmed, "[:=]", " ")
			local parts = {}
			for token in string.gmatch(cleanLine, "%S+") do
				table.insert(parts, token)
			end
			if #parts >= 2 then
				if string.lower(parts[2]) == target then
					return parts[1]
				elseif string.lower(parts[1]) == target then
					return parts[2]
				end
			elseif #parts == 1 then
				if not singleKeyFallback then
					singleKeyFallback = parts[1]
				end
			end
		end
	end
	return singleKeyFallback
end

					trimmed ~= ""
					and string.sub(trimmed, 1, 2) ~= "--"
					and string.sub(trimmed, 1, 1) ~= "#"

						table.insert(parts, token)

					then
						table.insert(lines, newEntry)
						replaced = true

						table.insert(lines, newEntry)

						table.insert(lines, trimmed)

	if not replaced then
		table.insert(lines, newEntry)
	end

	pcall(function()
		writefile(SAVE_FILE_NAME, finalContent)
	end)
	pcall(function()
		if writefile and LEGACY_SAVE_FILE then
			writefile(LEGACY_SAVE_FILE, cleanKey)
		end
	end)
	return true
end
local function DecryptSessionPayload(hexStr, keyStr)
	local keyBytes = {}
	for i = 1, #keyStr do
		table.insert(keyBytes, string.byte(keyStr, i))
	end
	if #keyBytes == 0 then
		return hexStr
	end
	local S = {}
	for i = 0, 255 do
		S[i] = i
	end
	local j = 0
	local kLen = #keyBytes
	for i = 0, 255 do
		j = (j + S[i] + keyBytes[(i % kLen) + 1]) % 256
		local temp = S[i]
		S[i] = S[j]
		S[j] = temp
	end
	local si, sj = 0, 0
	local chunks = {}
	local buf = {}
	local bufLen = 0
	local hexLen = #hexStr
	for i = 1, hexLen, 2 do
		local hexPair = string.sub(hexStr, i, i + 1)
		local encByte = tonumber(hexPair, 16) or 0
		si = (si + 1) % 256
		sj = (sj + S[si]) % 256
		local temp = S[si]
		S[si] = S[sj]
		S[sj] = temp
		local kByte = S[(S[si] + S[sj]) % 256]
		local decByte = (encByte - kByte) % 256
		bufLen = bufLen + 1
		buf[bufLen] = string.char(decByte)
		if bufLen >= 2048 then
			table.insert(chunks, table.concat(buf))
			buf = {}
			bufLen = 0
		end
	end
	if bufLen > 0 then
		table.insert(chunks, table.concat(buf))
	end
	return table.concat(chunks)
end

local function VerifyKeyWithServer(keyStr)
	if not keyStr or keyStr == "" then
		return false, { message = "Vui long nhap ma Key!" }
	end
	local cleanKey = string.gsub(keyStr, "^%s*(.-)%s*$", "%1")
	local hwid = GetDeviceHWID()
	local requestUrl = VERIFY_URL
		.. "&key="
		.. HttpService:UrlEncode(cleanKey)
		.. "&hwid="
		.. HttpService:UrlEncode(hwid)
	local ok, response, code = SafeHttpRequest(requestUrl)
	if ok and response and response ~= "" then
		local jsonOk, data = pcall(function()
			return HttpService:JSONDecode(response)
		end)
		if jsonOk and data then
			if data.valid then
				return true, data
			else
				return false, data
			end
		end
	end
	if code == 403 or (response and string.find(string.lower(tostring(response)), "403")) then
		return false, { message = "Key da duoc khoa cho thiet bi khac hoac script khac!" }
	elseif code == 400 or (response and string.find(string.lower(tostring(response)), "400")) then
		return false, { message = "Key da het han 12 gio hoac khong ton tai!" }
	end
	return false, { message = "Khong the ket noi den server AuraHub!" }
end

local isChecking = false
local function SetCheckButtonState(enabled, text)
	CheckKeyBtn.Active = enabled
	CheckKeyBtn.AutoButtonColor = enabled
	if not enabled then
		CheckKeyBtn.BackgroundColor3 = Color3.fromRGB(75, 60, 110)
		CheckKeyBtn.TextColor3 = Color3.fromRGB(180, 180, 200)
		CheckKeyBtn.Text = text or GetLang().loading_btn
	else
		CheckKeyBtn.BackgroundColor3 = Color3.fromRGB(124, 92, 255)
		CheckKeyBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
		CheckKeyBtn.Text = text or GetLang().check_key
	end
end
local function ProcessKeyVerification(keyToVerify, isAutoLogin)
	if isChecking then
		return
	end
	local cleanKey = string.gsub(keyToVerify or "", "^%s*(.-)%s*$", "%1")
	if not cleanKey or cleanKey == "" then
		StatusLabel.Text = GetLang().empty_key
		StatusLabel.TextColor3 = Color3.fromRGB(239, 68, 68)
		return
	end
	isChecking = true
	SetCheckButtonState(false, GetLang().loading_btn)
	StatusLabel.Text = GetLang().loading
	StatusLabel.TextColor3 = Color3.fromRGB(60, 224, 255)
	pcall(function()
		game:GetService("StarterGui"):SetCore("SendNotification", {
			Title = "⚡ AuraHub",
			Text = GetLang().loading,
			Duration = 4,
		})
	end)
	task.spawn(function()
		local valid, resData = VerifyKeyWithServer(cleanKey)
		if valid then
			local rem = (resData and (resData.time_remaining_human or "12h")) or "12h"
			SetCheckButtonState(false, GetLang().loading_btn)
			StatusLabel.Text = GetLang().loading
			StatusLabel.TextColor3 = Color3.fromRGB(60, 224, 255)
			SaveKeyToDevice(cleanKey, SCRIPT_NAME)
			LaunchProtectedScript(
				cleanKey,
				resData and resData.session_token,
				resData and resData.session_key,
				function(state, msg)
					if state == "loading" then
						StatusLabel.Text = GetLang().loading
						StatusLabel.TextColor3 = Color3.fromRGB(60, 224, 255)
						SetCheckButtonState(false, GetLang().loading_btn)
					elseif state == "error" then
						StatusLabel.Text = "✖ " .. tostring(msg)
						StatusLabel.TextColor3 = Color3.fromRGB(239, 68, 68)
						isChecking = false
						SetCheckButtonState(true, GetLang().check_key)
					elseif state == "success" then
						StatusLabel.Text = "✔ "
							.. (
								isVietnamese and "Khởi chạy script thành công!"
								or "Script launched successfully!"
							)
						StatusLabel.TextColor3 = Color3.fromRGB(34, 197, 94)
						task.wait(0.5)
						pcall(function()
							ScreenGui:Destroy()
						end)
					end
				end
			)
		else
			local errMsg = (resData and resData.message)
				or (isAutoLogin and GetLang().expired or GetLang().invalid)
			StatusLabel.Text = "✖ " .. tostring(errMsg)
			StatusLabel.TextColor3 = Color3.fromRGB(239, 68, 68)
			isChecking = false
			SetCheckButtonState(true, GetLang().check_key)
		end
	end)
end
CheckKeyBtn.MouseButton1Click:Connect(function()
	ProcessKeyVerification(KeyInput.Text, false)
end)
